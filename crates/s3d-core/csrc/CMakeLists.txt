cmake_minimum_required(VERSION 3.18)
project(s3d_manifold CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_MSVC_RUNTIME_LIBRARY)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
endif()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MANIFOLD_CROSS_SECTION OFF CACHE BOOL "" FORCE)
set(MANIFOLD_CBIND OFF CACHE BOOL "" FORCE)
set(MANIFOLD_PAR OFF CACHE BOOL "" FORCE)
set(MANIFOLD_TEST OFF CACHE BOOL "" FORCE)
set(MANIFOLD_PYBIND OFF CACHE BOOL "" FORCE)
set(MANIFOLD_DOWNLOADS OFF CACHE BOOL "" FORCE)
set(MANIFOLD_NO_IOSTREAM ON CACHE BOOL "" FORCE)

add_subdirectory(${MANIFOLD_SRC} manifold EXCLUDE_FROM_ALL)

add_library(s3d_manifold STATIC shim.cpp)
target_link_libraries(s3d_manifold PRIVATE manifold)

install(TARGETS s3d_manifold ARCHIVE DESTINATION lib)
install(FILES $<TARGET_FILE:manifold> DESTINATION lib)
